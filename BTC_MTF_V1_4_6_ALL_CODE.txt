def now_kst():
    return datetime.now(KST)


def ttl_get(key):
    with _LOCK:
        value = _CACHE.get(key)
        return value[1] if value and value[0] > time.monotonic() else None


def ttl_put(key, value, seconds):
    with _LOCK:
        if len(_CACHE) > 1600:
            _CACHE.clear()
        _CACHE[key] = (time.monotonic() + seconds, value)


def api_get(path, params=None):
    err = None
    for attempt in range(3):
        try:
            r = requests.get(API + path, params=params, timeout=(4, 13), headers={
                "Accept": "application/json", "User-Agent": "bithumb-btc-mtf-personal-v1/1.0"
            })
            if r.status_code in (429, 500, 502, 503, 504):
                raise RuntimeError(f"temporary HTTP {r.status_code}")
            r.raise_for_status()
            data = r.json()
            if not isinstance(data, list):
                raise RuntimeError("API array expected")
            return data
        except Exception as ex:
            err = ex
            if attempt < 2:
                time.sleep(.35 * (attempt + 1))
    raise RuntimeError(f"Bithumb {path}: {str(err)[:95]}")


def endpoint(tf):
    if tf not in TIMEFRAMES:
        raise ValueError("Unsupported timeframe")
    return "/v1/candles/" + ({"month": "months", "week": "weeks", "day": "days"}.get(tf, "minutes/" + str({"4h": 240, "1h": 60, "30m": 30, "15m": 15, "5m": 5, "3m": 3, "1m": 1}.get(tf, 5))))


def tf_end(start, tf):
    if tf == "month":
        m = start.month + 1
        y = start.year + (m > 12)
        m = (m - 1) % 12 + 1
        return start.replace(year=y, month=m, day=min(start.day, calendar.monthrange(y, m)[1]))
    if tf == "week":
        return start + timedelta(days=7)
    if tf == "day":
        return start + timedelta(days=1)
    return start + timedelta(minutes={"4h": 240, "1h": 60, "30m": 30, "15m": 15, "5m": 5, "3m": 3, "1m": 1}[tf])


def candles(market, tf, at=None, count=70):
    params = {"market": market, "count": min(200, count)}
    if at is not None:
        params["to"] = at.strftime("%Y-%m-%d %H:%M:%S")
    raw = api_get(endpoint(tf), params)
    clock = at or now_kst()
    result = {}
    for x in raw:
        try:
            start = datetime.fromisoformat(x["candle_date_time_kst"]).replace(tzinfo=KST)
            if tf_end(start, tf) > clock - timedelta(seconds=2):
                continue  # never use unfinished candle
            op, hi, lo, cl = [float(x[k]) for k in ("opening_price", "high_price", "low_price", "trade_price")]
            vol = float(x.get("candle_acc_trade_price") or 0)
            if not (math.isfinite(op) and math.isfinite(hi) and math.isfinite(lo) and math.isfinite(cl) and min(op, hi, lo, cl) > 0):
                continue
            result[start] = {"t": start, "o": op, "h": hi, "l": lo, "c": cl, "turnover": vol}
        except (KeyError, ValueError, TypeError, OverflowError):
            continue
    return sorted(result.values(), key=lambda r: r["t"])


def is_contiguous(rows, tf, n=10):
    if len(rows) < n:
        return False
    tail = rows[-n:]
    return all(tf_end(x["t"], tf) == y["t"] for x, y in zip(tail, tail[1:]))


def trend(rows, tf):
    if not is_contiguous(rows, tf, 10):
        return {"state": "unknown", "reason": "연속 마감봉 10개 부족", "bars": len(rows)}
    prices = [x["c"] for x in rows[-10:]]
    m5 = sum(prices[-5:]) / 5
    m10 = sum(prices) / 10
    c = prices[-1]
    state = "up" if c > m5 > m10 else "down" if c < m5 < m10 else "side"
    return {"state": state, "close": c, "ma5": m5, "ma10": m10,
            "closed_kst": tf_end(rows[-1]["t"], tf).isoformat(), "bars": len(rows)}


def aggregate(trends):
    states = [s for s in trends if s in ("up", "down", "side")]
    if len(states) < max(2, len(trends)//2):
        return "unknown"
    up, down = states.count("up"), states.count("down")
    return "up" if up > down and up >= (len(states)+1)//2 else "down" if down > up and down >= (len(states)+1)//2 else "side"


def historical_time(s):
    try:
        d = datetime.fromisoformat(str(s))
        if d.tzinfo is not None:
            raise ValueError("KST local date/time expected")
        d = d.replace(tzinfo=KST)
        if d.minute % 5 or d.second:
            raise ValueError("5분 단위 시각만 지원")
        if d > now_kst() - timedelta(minutes=245):
            raise ValueError("청산 240분이 지난 과거 시각만 검증 가능")
        if d < now_kst() - timedelta(days=360):
            raise ValueError("최근 360일 이내 시각 선택")
        return d
    except (TypeError, ValueError) as ex:
        raise ValueError(f"검증 시각 오류: {ex}")


def request_cfg(raw, need_market=False):
    if not isinstance(raw, dict):
        raise ValueError("요청 형식 오류")
    mode = raw.get("mode", "live")
    strategy = raw.get("strategy", "scalp")
    if strategy not in STRATEGIES:
        raise ValueError("단타/스윙 모드 오류")
    tf = raw.get("tf", "5m")
    if mode not in ("live", "backtest") or tf not in SCAN_TF:
        raise ValueError("모드 또는 시간봉 오류")
    fee = float(raw.get("fee", .04))
    if fee not in FEE_CHOICES:
        raise ValueError("편도 수수료는 0.04 또는 0.25만 선택")
    if mode == "backtest":
        at = historical_time(raw.get("at"))
    elif raw.get("snapshot_at"):
        try:
            at = datetime.fromisoformat(str(raw["snapshot_at"]).replace(" ", "T")).replace(tzinfo=KST)
            if abs((now_kst()-at).total_seconds()) > 900:
                raise ValueError("실시간 스캔 시각이 15분 이상 오래됨")
        except (ValueError, TypeError) as exc:
            raise ValueError(f"실시간 기준 시각 오류: {exc}")
    else:
        at = now_kst()
    market = raw.get("market") if need_market else None
    if mode == "backtest" and strategy == "swing" and at > now_kst() - timedelta(hours=121):
        raise ValueError("스윙은 과거 기준 시각 이후 120시간이 지나야 검증할 수 있어")
    if need_market and (not isinstance(market, str) or not MARKET_RE.fullmatch(market) or market.split("-", 1)[1] in STABLES):
        raise ValueError("KRW 코인 마켓만 조회 가능")
    return mode, tf, fee, at, market, strategy

def btc_context(cfg):
    mode, selected, fee, at, _, strategy = cfg
    key = ("btc", mode, selected, at.strftime("%Y-%m-%d %H:%M"))
    cached = ttl_get(key)
    if cached is not None:
        return cached
    states = {}
    last4 = None
    rows4 = []
    def run_tf(tf):
        try:
            # at is the snapshot clock; 1m candles are not *currently forming*.
            rs = candles("KRW-BTC", tf, at, count=48)
            return tf, trend(rs, tf), rs
        except Exception as exc:
            return tf, {"state": "unknown", "reason": str(exc)[:100]}, []
    with ThreadPoolExecutor(max_workers=5) as pool:
        for tf, state, rows in pool.map(run_tf, TIMEFRAMES):
            states[tf] = state
            if tf == selected:
                last4 = rows
            if tf == "4h":
                rows4 = rows
    lng = aggregate([states[tf]["state"] for tf in ("month", "week", "day", "4h")])
    sht = aggregate([states[tf]["state"] for tf in ("1h", "30m", "15m", "5m", "3m", "1m")])
    selected_return = round((last4[-1]["c"]/last4[-5]["c"]-1)*100, 4) if last4 and is_contiguous(last4, selected, 5) else None
    context = {"tf": states, "long": lng, "short": sht,
               "selected_return": selected_return, "selected_tf": selected,
               "asof_kst": at.strftime("%Y-%m-%d %H:%M:%S")}
    ttl_put(key, context, 3600 if mode == "backtest" else 900)
    ttl_put(("btc-h4",mode,at.strftime("%Y-%m-%d %H:%M")), rows4, 3600 if mode == "backtest" else 900)
    return context


def coin_metrics(rows, tf, btc_ret):
    if not is_contiguous(rows, tf, 23) or btc_ret is None:
        return None, "종목 23봉 연속 이력 또는 같은 봉 BTC 수익률 부족"
    closes = [x["c"] for x in rows]
    last, prev = rows[-1], rows[-2]
    m5 = sum(closes[-5:]) / 5
    m20 = sum(closes[-20:]) / 20
    prev6high = max(r["h"] for r in rows[-7:-1])
    breakout = (last["c"] / prev6high - 1) * 100
    change4 = (closes[-1]/closes[-5]-1)*100
    relative = change4-btc_ret
    volprev = sum(x["turnover"] for x in rows[-21:-1])/20
    volratio = last["turnover"]/volprev if volprev>0 else 0
    body = (last["c"]-last["o"])/max(last["h"]-last["l"], 1e-14)
    tr = [max(x["h"]-x["l"], abs(x["h"]-y["c"]), abs(x["l"]-y["c"])) for x,y in zip(rows[-14:], rows[-15:-1])]
    atr = sum(tr)/len(tr)/last["c"]*100 if tr else None
    return {"close": last["c"], "ma5": m5, "ma20": m20,
            "breakout_pct": round(breakout, 4), "coin_return_pct": round(change4, 4),
            "btc_return_pct": btc_ret, "relative_pct": round(relative, 4),
            "volume_ratio": round(volratio, 3), "body_ratio": round(body, 3),
            "atr_pct": round(atr, 3) if atr is not None else None,
            "closed_kst": tf_end(last["t"], tf).strftime("%Y-%m-%d %H:%M")}, None


def coin_signal(metrics, ctx, fee):
    m = metrics
    conditions = {
        "trend": m["close"] > m["ma5"] > m["ma20"],
        "breakout": m["breakout_pct"] > 0,
        "relative": m["relative_pct"] >= .5,
        "volume": m["volume_ratio"] >= .7,
        "candle": m["body_ratio"] >= .5,
    }
    bear = ctx["long"] == "down" and ctx["short"] == "down"
    independent = m["relative_pct"] >= 1.5 and m["breakout_pct"] >= .25
    buy = all(conditions.values()) and (not bear or independent)
    watch = (conditions["trend"] or conditions["breakout"]) and m["relative_pct"] > 0 and m["body_ratio"] > 0
    if not buy and not watch:
        return None
    kind = "BUY" if buy else "WATCH"
    causes = []
    causes.append(f"같은 {ctx['selected_tf']} 기준 BTC 대비 상대강도 {m['relative_pct']:+.2f}%p (코인 {m['coin_return_pct']:+.2f}% / BTC {m['btc_return_pct']:+.2f}%)")
    causes.append(f"직전 6봉 고점 대비 마감 돌파폭 {m['breakout_pct']:+.2f}% · 거래대금 배수 {m['volume_ratio']:.2f}배")
    causes.append(f"종가 > MA5 > MA20 {'충족' if conditions['trend'] else '미충족'} · 양봉 몸통 비중 {m['body_ratio']*100:.0f}%")
    causes.append(f"BTC 장기 {state_ko(ctx['long'])}, 단기 {state_ko(ctx['short'])}" + (" · 약세장 상대강도 강화 충족" if bear and independent else ""))
    if kind == "WATCH":
        missing = [k for k, ok in conditions.items() if not ok]
        names = {"trend": "상승 배열", "breakout": "직전 고점 돌파", "relative": "BTC 대비 0.5%p 우위", "volume": "거래대금 0.7배", "candle": "양봉 몸통 50%"}
        causes.append("BUY 보류: " + (", ".join(names[k] for k in missing) if missing else "BTC 동시 강한 하락 중 독립 상승 강도 부족"))
    else:
        causes.append("모든 진입 품질 조건 충족 · 실제 주문이 아닌 연구용 BUY 후보")
    # Exchange fee applies per leg; backtest uses the same target levels.
    gross_tp1 = ((1+.01)*(1+fee/100)/(1-fee/100)-1)*100
    gross_tp2 = ((1+.10)*(1+fee/100)/(1-fee/100)-1)*100
    return {"signal": kind, "metrics": m, "explanation": causes,
            "plan": {"tp1_net": 1.0, "tp1_fraction": .7, "tp2_net": 10.0,
                     "tp2_fraction": .3, "stop_gross": -2.0, "max_minutes": 240,
                     "tp1_gross_pct": round(gross_tp1, 3), "tp2_gross_pct": round(gross_tp2, 3)}}


def state_ko(s):
    return {"up": "상승", "side": "횡보", "down": "하락", "unknown": "자료 부족"}.get(s, "자료 부족")


def backtest_exit(market, at, fee, getter=candles):
    """Simulate one snapshot entry at the 5m open and 240m observed path.

    Both stop and TP within a 5-minute candle: conservatively stop first.
    Prior 5-minute candles are not traded, and no synthetic candles are made.
    """
    out = {"status": "UNKNOWN", "net_pct": None, "reason": "", "tp1": False, "tp2": False,
           "stop": False, "minutes": None, "ambiguous": False}
    try:
        rows = getter(market, "5m", at+timedelta(minutes=241), count=100)
    except Exception as ex:
        return {**out, "reason": "청산 경로 조회 실패: " + str(ex)[:80]}
    indexed = {r["t"]:r for r in rows}
    if any(at+timedelta(minutes=5*k) not in indexed for k in range(48)):
        return {**out, "reason": "240분 중 5분봉 누락: 미래 가격 경로 확인 불가"}
    entry = indexed[at]["o"]
    if entry <= 0:
        return {**out, "reason": "진입 시가 없음"}
    cost = fee / 100
    p1 = entry*(1.01)*(1+cost)/(1-cost)
    p2 = entry*(1.10)*(1+cost)/(1-cost)
    stop = entry*.98
    fraction, realized = 1.0, 0.0
    tp1 = tp2 = False
    def net(px):
        return (px*(1-cost)/(entry*(1+cost))-1)*100
    for k in range(48):
        b = indexed[at+timedelta(minutes=5*k)]
        op, hi, lo, cl = b["o"], b["h"], b["l"], b["c"]
        if lo <= stop:
            ambiguous = (hi >= p1 if not tp1 else hi >= p2)
            return {**out, "status": "RESOLVED", "net_pct": round(realized + fraction*net(min(op,stop)), 6),
                    "reason": "", "tp1": tp1, "tp2": tp2, "stop": True,
                    "minutes": (k+1)*5, "ambiguous": ambiguous}
        if not tp1 and hi >= p1:
            realized += .7*net(p1)
            fraction = .3
            tp1 = True
        if tp1 and hi >= p2:
            realized += fraction*net(p2)
            return {**out, "status": "RESOLVED", "net_pct": round(realized, 6),
                    "reason": "", "tp1": True, "tp2": True, "stop": False,
                    "minutes": (k+1)*5, "ambiguous": False}
        if k == 47:
            return {**out, "status": "RESOLVED", "net_pct": round(realized + fraction*net(cl), 6),
                    "reason": "", "tp1": tp1, "tp2": False, "stop": False,
                    "minutes": 240, "ambiguous": False}
    return {**out, "reason": "결과 확인 불가"}


def _avg(values):
    return sum(values)/len(values)


def swing_metrics(day, h4, h1, btc4, at):
    """Only completed candles at the snapshot and synchronized BTC 4h series."""
    # Historical outage gaps should not wipe out the entire market. Do not
    # fabricate missing candles: the indicator module validates real-bar
    # density and blocks BUY while recent hourly data is discontinuous.
    if not (len(day)>=24 and len(h4)>=34 and len(h1)>=16):
        return None, "스윙 판정용 실제 캔들 수 부족(일 24, 4시간 34, 1시간 16)"
    if at-tf_end(h4[-1]["t"],"4h") > timedelta(hours=8) or at-tf_end(h1[-1]["t"],"1h") > timedelta(hours=3):
        return None, "4시간/1시간 최신 완료봉이 오래됨"
    btcmap={b["t"]:b["c"] for b in btc4}
    if h4[-1]["t"] not in btcmap or h4[-5]["t"] not in btcmap:
        return None, "BTC와 코인의 동일 시각 4시간봉 부족"
    d=[x["c"] for x in day]; f=[x["c"] for x in h4]; hour=[x["c"] for x in h1]
    day_ma20=_avg(d[-20:]); day_prev20=_avg(d[-24:-4])
    ma7=_avg(f[-7:]); ma14=_avg(f[-14:]); ma30=_avg(f[-30:]); prev7=_avg(f[-10:-3])
    h1ma7=_avg(hour[-7:]); h1ma14=_avg(hour[-14:])
    ret16=(f[-1]/f[-5]-1)*100; btc16=(btcmap[h4[-1]["t"]]/btcmap[h4[-5]["t"]]-1)*100
    prev12high=max(x["h"] for x in h4[-13:-1]); high6=max(x["h"] for x in h4[-7:-1])
    lo_recent=min(x["l"] for x in h4[-7:-1]); lo_prior=min(x["l"] for x in h4[-13:-7])
    volbase=_avg([x["turnover"] for x in h4[-21:-1]]); ratio=h4[-1]["turnover"]/volbase if volbase else 0
    tr=[max(x["h"]-x["l"],abs(x["h"]-y["c"]),abs(x["l"]-y["c"])) for x,y in zip(h4[-14:],h4[-15:-1])]
    atr=_avg(tr)/f[-1]*100
    dayrise=(d[-1]/d[-2]-1)*100; ret24=(f[-1]/f[-7]-1)*100
    ext=(f[-1]/ma7-1)*100; support=lo_recent*.998
    risk=(f[-1]/support-1)*100
    breakout=(f[-1]/prev12high-1)*100
    return {
      "close":f[-1],"ma5":ma7,"ma20":ma14,"day_ma20":round(day_ma20,6),"h4_ma30":round(ma30,6),
      "coin_return_pct":round(ret16,4),"btc_return_pct":round(btc16,4),"relative_pct":round(ret16-btc16,4),
      "breakout_pct":round(breakout,4),"volume_ratio":round(ratio,3),"atr_pct":round(atr,3),
      "h4_distance_ma7_pct":round(ext,3),"structure_risk_pct":round(risk,3),"support":round(support,8),
      "day_rise_pct":round(dayrise,3),"rise24_pct":round(ret24,3),
      "h4_higher_low":lo_recent>lo_prior,"h4_higher_high":high6>max(x["h"] for x in h4[-13:-7]),
      "day_up":d[-1]>day_ma20 and day_ma20>day_prev20,
      "h4_up":f[-1]>ma7>ma14 and ma7>prev7,
      "h1_up":hour[-1]>h1ma7>h1ma14,
      "recovery":min(x["l"] for x in h4[-3:-1])<=ma7*1.016 and f[-1]>f[-2],
      "closed_kst":tf_end(h4[-1]["t"],"4h").strftime("%Y-%m-%d %H:%M"),

def strict_swing_gate(m, ctx, day, h4):
    """V1.4 research gate; never uses candles after the snapshot."""
    h4_ma20 = sum(x["c"] for x in h4[-20:]) / 20
    h4_ma20_prev = sum(x["c"] for x in h4[-21:-1]) / 20
    checks = {
        "점수 9/10 이상": m["score"] >= 9,
        "일봉 MA20 상향 + 22봉 연속": m["day_up"] and is_contiguous(day, "day", 22),
        "4시간봉 MA20 상회·상향 + 21봉 연속": (
            is_contiguous(h4, "4h", 21) and m["close"] > h4_ma20 > h4_ma20_prev),
        "4시간봉 MA7/14 상승 배열": m["h4_up"],
        "1시간봉 반등 확인": m["checks"]["h1_confirm"],
        "RSI(14) 50 이상": m["rsi14"] >= 50,
        "MACD 히스토그램 양수": m["macd_hist"] > 0,
        "4시간 MACD 골든 상태 (실제 교차와 구분)": m["macd_h4"]["state"] == "GOLDEN",
        "4시간 거래대금 1.1배 이상": m["volume_ratio"] >= 1.1,
        "BTC 장·단기 비하락": (
            ctx["long"] in ("up", "side") and ctx["short"] in ("up", "side")),
        "최근 봉 공백 없음": not m.get("data_gap_recent", True),
        "저점·지지 9% 내 / 선택 시간봉 확인": (
            1.2 <= m["rebound_from_low_pct"] <= 9
            and 0 <= m["structure_risk_pct"] <= 9
            and m.get("timing_ok", False)),
    }
    return {
        "passed": all(checks.values()), "checks": checks,
        "missing": [k for k, v in checks.items() if not v],
        "h4_ma20": round(h4_ma20, 8),
        "h4_ma20_prev": round(h4_ma20_prev, 8)
    }


def swing_signal(m,ctx,fee):
    rel=m["relative_pct"];bear=ctx["long"]=="down" or ctx["short"]=="down"
    strong_rel=rel>=(2.0 if bear else 1.0)
    excessive=m["h4_distance_ma7_pct"]>7.5 or m["rise24_pct"]>22.0 or m["structure_risk_pct"]>12
    setup=m["breakout_pct"]>0 or m["recovery"]
    good=(m["day_up"] and m["h4_up"] and m["h1_up"] and strong_rel
          and m["h4_higher_low"] and setup and m["volume_ratio"]>=.7
          and m["structure_risk_pct"]<=8.0 and m.get("timing_ok",True) and not excessive)
    watch=((m["day_up"] or m["h4_up"]) and rel>=.3 and
           (m["h4_up"] or m["h4_higher_high"] or m["breakout_pct"]>0))
    if not good and not watch:return None
    kind="BUY" if good else "WATCH"
    checks={
      "일봉 MA20 상회·우상향":m["day_up"],
      "4시간봉 MA7·MA14 상승":m["h4_up"],
      "4시간봉 저점 상승":m["h4_higher_low"],
      "1시간봉 재상승":m["h1_up"],
      "BTC 대비 강도 충족":strong_rel,
      "4시간봉 돌파·눌림 회복":setup,
      "4시간봉 거래대금 0.7배 이상":m["volume_ratio"]>=.7,
      "구조적 손절 거리 8% 이내":m["structure_risk_pct"]<=8.0,
      "선택한 검색봉 상승 확인":m.get("timing_ok",True),
      "급등 추격 위험 낮음":not excessive,
    }
    causes=[
      f"스윙: 일봉 {'상승' if m['day_up'] else '미확정'} · 4시간봉 {'상승' if m['h4_up'] else '미확정'} · 1시간봉 {'상승' if m['h1_up'] else '미확정'}",
      f"동일한 16시간 동안 코인 {m['coin_return_pct']:+.2f}%, BTC {m['btc_return_pct']:+.2f}% → 상대강도 {rel:+.2f}%p",
      f"4시간봉 고점 돌파폭 {m['breakout_pct']:+.2f}% · 최근 고점/저점 상승 {m['h4_higher_high']}/{m['h4_higher_low']}",
      f"4시간봉 MA7 이격 {m['h4_distance_ma7_pct']:+.2f}% · 최근 24시간 {m['rise24_pct']:+.2f}% · 구조적 지지까지 {m['structure_risk_pct']:.2f}%",
      f"4시간봉 거래대금 배수 {m['volume_ratio']:.2f}배 · BTC 장기 {state_ko(ctx['long'])}, 단기 {state_ko(ctx['short'])}",
      f"선택 검색봉 {m.get('selected_tf','4h')} 추세 {'상승' if m.get('timing_ok',True) else '미확정'}",
    ]
    if kind=="WATCH":
        missing=[k for k,v in checks.items() if not v]
        causes.append("BUY 보류: "+(", ".join(missing) if missing else "상승 확인 대기"))
    else:
        causes.append("스윙 구조 조건 충족 · 체결 전 호가/집중보유/공시 위험 별도 확인")
    if excessive:causes.append("⚠ 급등 추격 위험: 이미 많이 오른 시점이므로 신규 진입은 WATCH")
    cost=fee/100
    tp1=((1.05)*(1+cost)/(1-cost)-1)*100
    tp2=((1.12)*(1+cost)/(1-cost)-1)*100
    return {"signal":kind,"metrics":m,"explanation":causes,"plan":{
      "tp1_net":5.0,"tp1_fraction":.5,"tp2_net":12.0,"tp2_fraction":.5,
      "stop_gross":-5.0,"max_minutes":7200,
      "tp1_gross_pct":round(tp1,3),"tp2_gross_pct":round(tp2,3),
      "support":m["support"],"support_risk_pct":m["structure_risk_pct"]}}


def swing_backtest_exit(market, at, fee, getter=candles, detailed=False):
    """Simulate 120h only on verified Bithumb 1h candles.

    V1.4.6 targeted recheck is triggered only when the replay actually
    encounters a missing hour. Do NOT fill missing hours with prices from
    30m or carry-forward data. Existing exit-order and fee logic unchanged.
    """
    out = {"status": "UNKNOWN", "net_pct": None, "reason": "", "tp1": False,
           "tp2": False, "stop": False, "minutes": None, "ambiguous": False}
    entry_at = at.replace(minute=0, second=0, microsecond=0)
    if entry_at < at:
        entry_at += timedelta(hours=1)
    path = HourlyReplayPath(market, entry_at, getter, detailed=detailed)
    def done(**fields):
        return {**out, **fields, "audit": path.audit()}

    first = path.hour(entry_at)
    if first is None or first.get("o", 0) <= 0:
        return done(reason="진입 시각 "+entry_at.strftime("%Y-%m-%d %H:%M")
                    +" 1시간봉 미확인 (재조회: "+path.audit()["status_code"]+")",
                    reason_code="ENTRY_HOUR_MISSING")
    entry = first["o"]
    cost = fee / 100
    p1 = entry * 1.05 * (1+cost)/(1-cost)
    p2 = entry * 1.12 * (1+cost)/(1-cost)
    stop = entry * .95
    rest, realized, tp1 = 1.0, 0.0, False
    def net(px):
        return (px*(1-cost)/(entry*(1+cost))-1)*100
    for i in range(120):
        r = path.hour(entry_at + timedelta(hours=i))
        if r is None:
            clock = (entry_at + timedelta(hours=i)).strftime("%Y-%m-%d %H:%M")
            code = path.audit()["status_code"]
            return done(reason=(f"진입 후 {i}시간째 {clock} KST 1시간봉 미확인"
                                +f" (재조회 {code}); 이후 청산 미확정"),
                        reason_code="MISSING_HOURLY_BAR",
                        tp1=tp1, minutes=i*60)
        if r["l"] <= stop:
            return done(status="RESOLVED",
                        net_pct=round(realized + rest*net(min(r["o"], stop)), 6),
                        tp1=tp1, stop=True, minutes=(i+1)*60,
                        ambiguous=r["h"] >= (p2 if tp1 else p1))
        if not tp1 and r["h"] >= p1:
            tp1 = True
            realized += .5 * net(p1)
            rest = .5
        if tp1 and r["h"] >= p2:
            return done(status="RESOLVED",
                        net_pct=round(realized + rest*net(p2), 6),
                        tp1=True, tp2=True, minutes=(i+1)*60)
        if i == 119:
            return done(status="RESOLVED",
                        net_pct=round(realized + rest*net(r["c"]), 6),
                        tp1=tp1, minutes=7200)
    return done(reason="미확정 경로", reason_code="UNKNOWN_PATH")

def _valid_turnover(row):
    try:
        v = float(row["turnover"])
    except (KeyError, ValueError, TypeError, OverflowError):
        return None
    return v if math.isfinite(v) and v >= 0 else None


def snapshot_liquidity(h1, snapshot):
    """Read-only 24h historical turnover / hourly completeness from real bars.

    Note: missing hourly candles are NOT treated as zero-turnover evidence.
    A present hourly candle can still represent a thin market.
    """
    floor = snapshot.replace(minute=0, second=0, microsecond=0)
    start = floor - timedelta(hours=24)
    expected = [start + timedelta(hours=i) for i in range(24)]
    index = {}
    for r in h1:
        try:
            stamp = r["t"]
        except (KeyError, TypeError):
            continue
        if stamp not in expected or stamp in index:
            continue
        if stamp + timedelta(hours=1) > snapshot:
            continue
        amount = _valid_turnover(r)
        if amount is None:
            continue
        index[stamp] = amount

    present = len(index)
    recent6 = sum(t in index for t in expected[-6:])
    missing = [t for t in expected if t not in index]

def investigate_missing_hour(market, missing_hour, getter):

    cursor = end + timedelta(minutes=1)
    observations = {}
    errors = {}
    samples = {}
    for tf, count in (("5m", 15), ("1m", 70)):
        try:
            items = getter(market, tf, cursor, count=count)
            interval = timedelta(minutes=5 if tf == "5m" else 1)

                and (row["t"] + interval) <= cursor

"""BTC MTF V1.4.5: bounded, auditable candle recovery for 120h replay.

No synthetic prices are created. The replay is based only on real Bithumb
1h OHLC candles, with exact KST timestamps. A 30m query is a diagnostic
only and NEVER supplies a missing 1h price for virtual exit execution.
"""
from datetime import timedelta

AUDIT_RULE = "REPLAY_V145"
LIMIT_HOURS = 120


def _clock(dt):
    return dt.strftime("%Y-%m-%d %H:%M") if dt else ""


class HourlyReplayPath:
    def __init__(self, market, first_hour, getter, detailed=False, limit=LIMIT_HOURS):
        self.market = market
        self.first_hour = first_hour
        self.end_hour = first_hour + timedelta(hours=limit)
        self.getter = getter
        self.detailed = bool(detailed)
        self.index = {}
        self.api_calls = 0
        self.hourly_rechecks = 0
        self.half_hour_diagnostics = 0
        self.recovered = 0
        self.initial_error = ""
        self.initial_missing_count = None
        self.initial_missing_samples = []
        self.last_gap_at = None
        self.last_gap_status = "NONE"
        self.last_lower_tf_count = None
        self.tried_times = set()
        self.max_hourly_rechecks = 2
        self._initial_load()

    def _call(self, tf, cursor, count):
        self.api_calls += 1
        return self.getter(self.market, tf, cursor, count=count)

    def _merge(self, rows):
        found = 0
        for row in rows:
            if not isinstance(row, dict) or "t" not in row:
                continue
            t = row["t"]
            if not self.first_hour <= t < self.end_hour:
                continue
            # Never overwrite a verified first occurrence with a conflicting
            # response from another request.
            if t not in self.index:
                self.index[t] = row
                found += 1
        return found

    def _initial_load(self):
        # Count=160 is preserved from V1.4.4, so the primary baseline is
        # directly comparable. "to" is KST and exclusive in Bithumb docs.
        cursor = self.end_hour + timedelta(minutes=1)
        try:
            rows = self._call("1h", cursor, 160)
            self._merge(rows)
        except Exception as ex:
            self.initial_error = str(ex)[:120]
        missing = [
            self.first_hour + timedelta(hours=i)
            for i in range(LIMIT_HOURS)
            if self.first_hour + timedelta(hours=i) not in self.index
        ]
        self.initial_missing_count = len(missing)
        self.initial_missing_samples = [_clock(x) for x in missing[:5]]

    def hour(self, stamp):
        """Return verified 1h candle, or None if source cannot confirm it."""
        row = self.index.get(stamp)
        if row is not None:
            return row
        self.last_gap_at = stamp
        if stamp in self.tried_times:
            return None
        self.tried_times.add(stamp)
        if self.hourly_rechecks >= self.max_hourly_rechecks:
            self.last_gap_status = "RECHECK_BUDGET_EXHAUSTED"
            return None

        # A small exact time-bounded request avoids relying on the ranking
        # of sparse candles from a single large historical batch.
        self.hourly_rechecks += 1
        try:
            fresh = self._call("1h", stamp + timedelta(hours=1, minutes=1), 8)
            self._merge(fresh)
        except Exception as ex:
            self.last_gap_status = "HOURLY_RECHECK_API_ERROR"
            self.last_error = str(ex)[:120]
            return None
        row = self.index.get(stamp)
        if row is not None:
            self.recovered += 1
            self.last_gap_status = "RECOVERED_1H"
            return row

        self.last_gap_status = "SOURCE_1H_NOT_RETURNED"
        if not self.detailed:
            return None

        # A 30m source check can help distinguish a suspicious missing 1h
        # candle from a likely low-activity period. Never aggregate 30m
        # into 1h OHLC or assume no trade occurred solely from this response.
        self.half_hour_diagnostics += 1
        try:
            rows30 = self._call("30m", stamp + timedelta(hours=1, minutes=1), 8)
            self.last_lower_tf_count = sum(
                1 for r in rows30
                if isinstance(r, dict) and "t" in r
                and stamp <= r["t"] < stamp + timedelta(hours=1)
            )
            if self.last_lower_tf_count:
                self.last_gap_status = "SOURCE_1H_MISSING_BUT_30M_EXISTS"
            else:
                self.last_gap_status = "SOURCE_1H_30M_NOT_RETURNED"
        except Exception as ex:
            self.last_gap_status = "SOURCE_1H_MISSING_30M_API_ERROR"
            self.last_error = str(ex)[:120]
        return None

    def audit(self):
        code = self.last_gap_status
        if code == "NONE":
            code = ("INITIAL_API_ERROR" if self.initial_error else
                    "COMPLETE_1H_PATH" if self.initial_missing_count == 0
                    else "NO_RECHECK_NEEDED_BEFORE_EXIT")
        return {
            "rule": AUDIT_RULE,
            "status_code": code,
            "initial_api_error": self.initial_error,
            "detail_error": getattr(self, "last_error", ""),
            "initial_missing_hours": self.initial_missing_count,
            "initial_missing_samples_kst": list(self.initial_missing_samples),
            "verified_hourly_candles": len(self.index),
            "first_unresolved_or_recovered_hour_kst": _clock(self.last_gap_at),
            "hourly_recheck_calls": self.hourly_rechecks,
            "half_hour_diagnostic_calls": self.half_hour_diagnostics,
            "total_candle_api_calls": self.api_calls,
            "recovered_one_hour_candles": self.recovered,
            "lower_timeframe_bars_in_missing_hour": self.last_lower_tf_count,
            "source_limitations": (
                "UNKNOWN remains UNKNOWN unless exact Bithumb 1h candle was returned. "
                "30m is diagnosis only, not synthetic execution prices."
            ),
        }

"""Compare MACD line/signal position with an actual completed-bar crossover.

Standard 4h/day MACD: 12,26,9. Chart-compatible daily variant: 3,10,16.
Cross event refers ONLY to the latest completed candle. Bullish status does
not imply that a new golden cross happened on that candle.
"""
from datetime import timedelta
from btc_mtf_rebound_v12 import ema_series


def analyze_macd(rows, fast, slow, signal, tf, bar_end, contiguous):
    minimum = slow + signal + 10
    empty = {
        "state": "UNKNOWN", "event": "NONE", "last_event": "NONE",
        "last_event_kst": None, "bars_since_cross": None,
        "macd": None, "signal": None, "hist": None, "prev_hist": None,
        "continuous": False, "params": f"{fast},{slow},{signal}",
        "tf": tf, "closed_kst": None,
    }
    if not (0 < fast < slow and signal > 0 and len(rows) >= minimum):
        return empty
    prices = [r["c"] for r in rows]
    efast = ema_series(prices, fast)
    eslow = ema_series(prices, slow)
    line = [x-y for x, y in zip(efast, eslow)]
    sig = ema_series(line, signal)
    hist = [x-y for x, y in zip(line, sig)]
    recent_ok = contiguous(rows, tf, 3)
    current = hist[-1]
    state = "GOLDEN" if current > 0 else "DEAD" if current < 0 else "FLAT"
    if not recent_ok:
        state = "UNKNOWN"
    def cross(previous, latest):
        if previous <= 0 < latest:
            return "GOLDEN_CROSS"
        if previous >= 0 > latest:
            return "DEAD_CROSS"
        return "NONE"
    event = cross(hist[-2], hist[-1]) if recent_ok else "NONE"
    last_event = "NONE"
    last_at = None
    bars_ago = None
    # Search last 24 completed candles, but never claim a crossover across
    # an interval for which a real bar is missing.
    step = timedelta(hours=4) if tf == "4h" else timedelta(days=1)
    for i in range(len(rows)-1, max(0, len(rows)-25), -1):
        if rows[i]["t"] - rows[i-1]["t"] != step:
            continue
        kind = cross(hist[i-1], hist[i])
        if kind != "NONE":
            last_event = kind
            last_at = bar_end(rows[i]["t"], tf).strftime("%Y-%m-%d %H:%M")
            bars_ago = len(rows)-1-i
            break
    return {
        "state": state, "event": event,
        "last_event": last_event, "last_event_kst": last_at,
        "bars_since_cross": bars_ago, "continuous": recent_ok,
        "macd": round(line[-1], 10), "signal": round(sig[-1], 10),
        "hist": round(current, 10), "prev_hist": round(hist[-2], 10),
        "params": f"{fast},{slow},{signal}", "tf": tf,
        "closed_kst": bar_end(rows[-1]["t"], tf).strftime("%Y-%m-%d %H:%M"),
    }


def all_macd_views(day, h4, bar_end, contiguous):
    return {
        "h4": analyze_macd(h4, 12, 26, 9, "4h", bar_end, contiguous),
        "day": analyze_macd(day, 12, 26, 9, "day", bar_end, contiguous),
        "day_chart": analyze_macd(day, 3, 10, 16, "day", bar_end, contiguous),
    }

"""BTC MTF V1.3 | quality-weighted early rebound classification.
Keeps V1.2's 7 completed-candle technical checks and adds three independent
quality checks. A score is a research heuristic, not a probability of profit.
"""
from btc_mtf_rebound_v12 import bottom_rebound_metrics as _metrics
from btc_mtf_rebound_v12 import rebound_signal as _legacy_signal

RULE_VERSION = "REBOUND_V1.3_SCORE10"
MAX_SCORE = 10

def bottom_rebound_metrics(*args, **kwargs):
    m, issue = _metrics(*args, **kwargs)
    if m is None:
        return m, issue
    quality = {
        "btc_strength": m["relative_pct"] >= 0.5,
        "entry_zone": 1.2 <= m["rebound_from_low_pct"] <= 11.0 and m["rise24_pct"] <= 12,
        "risk_control": 0 <= m["structure_risk_pct"] <= 11,
    }
    m["technical_score"] = m["score"]
    m["quality_checks"] = quality
    m["quality_score"] = sum(quality.values())
    m["score"] = m["technical_score"] + m["quality_score"]
    m["score_total"] = MAX_SCORE
    m["score_rule"] = RULE_VERSION
    return m, None

def rebound_signal(m, ctx, fee):
    if not m:
        return None
    c = m["checks"]
    q = m["quality_checks"]
    from_low = m["rebound_from_low_pct"]
    risk = m["structure_risk_pct"]
    rel = m["relative_pct"]
    bearish = ctx["long"] == "down" or ctx["short"] == "down"
    relative_floor = 0.8 if bearish else 0.5

    # Mandatory prefilters: scores never excuse bad entry location or
    # relative weakness. Already-exploded coins are excluded, not WATCH.
    watch_eligible = (
        m["score"] >= 7
        and m["technical_score"] >= 4
        and rel >= relative_floor
        and 1.2 <= from_low <= 11
        and 0 <= risk <= 11
        and m["rise24_pct"] <= 12
        and 0.15 <= m["bb_position"] <= 1.0
        and m["volume_ratio"] >= 0.35
        and m["rsi14"] <= 73
    )
    if not watch_eligible:
        return None

    trigger = (
        c["h1_confirm"] and c["ma_recover"]
        and c["rsi_recover"] and c["macd_recover"]
        and (c["bb_recover"] or c["ichimoku_recover"])
    )
    buy = (
        m["score"] >= 8 and trigger and rel >= relative_floor
        and m["volume_ratio"] >= 0.7
        and from_low <= 9.0 and risk <= 9.0
        and m["bb_position"] <= 0.95
        and m.get("timing_ok", True)
        and not m.get("data_gap_recent", False)  # avoid BUY on interrupted 1h/4h history
    )
    kind = "BUY" if buy else "WATCH"
    stage = ("반등 확인 BUY" if buy else
             "점검 공백 · BUY 보류" if m.get("data_gap_recent", False) else
             "반등 직전 WATCH" if m["score"] >= 8 else "반등 준비 WATCH")
    legacy = dict(m)
    legacy["score"] = m["technical_score"]  # preserve legacy /7 explanation and plan
    raw = _legacy_signal(legacy, ctx, fee)
    if raw is None:
        return None

    names = {
        "bb_recover": "볼린저밴드 회복", "ma_recover": "MA5 기울기 회복",
        "ichimoku_recover": "일목 전환선", "rsi_recover": "RSI 상승",
        "macd_recover": "MACD 모멘텀", "volume_recover": "4시간 거래대금",
        "h1_confirm": "1시간봉 반등"
    }
    quality_names = {
        "btc_strength": "BTC 대비 +0.5%p 이상",
        "entry_zone": "저점에서 +1.2~11% 초기 구간",
        "risk_control": "저점 지지까지 11% 이내"
    }
    tech_details = " · ".join(("✓ " if passed else "× ") + names[k] for k, passed in c.items())
    qual_details = " · ".join(("✓ " if passed else "× ") + quality_names[k] for k, passed in q.items())
    why = [
        f"반등 점수 {m['score']}/10 = 지표 {m['technical_score']}/7 + 진입 품질 {m['quality_score']}/3",
        f"판정 {stage} · 10점이어도 반등 확정이 아니며 BUY 필수 확인 조건 적용",
        "기술 지표: " + tech_details,
        "진입 품질: " + qual_details,
        ("⚠ 최근 1시간·4시간 마감봉에 공백이 있어 BUY를 자동 보류함. 실거래 캔들만 이용하며 누락봉 보간 없음."
         if m.get("data_gap_recent") else "최근 1시간·4시간 마감봉 연속성 확인"),
        *raw["explanation"][1:7],
        f"저점 대비 +{from_low:.2f}% · 저점 기준 위험 거리 {risk:.2f}% · 24시간 등락 {m['rise24_pct']:+.2f}%",
    ]
    if not buy:
        missing = [names[k] for k in ("h1_confirm", "ma_recover", "rsi_recover", "macd_recover") if not c[k]]
        if not (c["bb_recover"] or c["ichimoku_recover"]):

"""BTC MTF V1.2 | early swing rebound indicators using completed OHLC only."""
from __future__ import annotations
from math import sqrt
from datetime import timedelta

RULE_VERSION = "REBOUND_V1.2"

def average(xs): return sum(xs)/len(xs)

def ema_series(xs,period):
    a=2/(period+1);v=None;out=[]
    for p in xs:
        v=p if v is None else v+a*(p-v)
        out.append(v)
    return out

def rsi_series(closes,length=14):
    if len(closes)<=length:return []
    gains=[max(y-x,0) for x,y in zip(closes,closes[1:])]
    losses=[max(x-y,0) for x,y in zip(closes,closes[1:])]
    up=average(gains[:length]);dn=average(losses[:length])
    def rsi(u,d):
        return 50.0 if u==d==0 else (100.0 if d==0 else 100-100/(1+u/d))
    out=[rsi(up,dn)]
    for u,d in zip(gains[length:],losses[length:]):
        up=(up*(length-1)+u)/length;dn=(dn*(length-1)+d)/length
        out.append(rsi(up,dn))
    return out

def macd_hist(closes):
    e12,e26=ema_series(closes,12),ema_series(closes,26)
    macd=[a-b for a,b in zip(e12,e26)]
    sig=ema_series(macd,9)
    return [m-s for m,s in zip(macd,sig)],macd[-1],sig[-1]

def bollinger(closes,index=None):
    i=len(closes)-1 if index is None else index
    xs=closes[i-19:i+1]
    mid=average(xs);sd=sqrt(average([(v-mid)**2 for v in xs]))
    width=max(4*sd,mid*1e-8)
    return mid,mid-2*sd,mid+2*sd,(closes[i]-(mid-2*sd))/width

def midpoint(tail):
    return (max(x['h'] for x in tail)+min(x['l'] for x in tail))/2

def ichimoku(rows):
    # Span at current candle = computed 26 completed bars earlier, no future data.
    tenkan=midpoint(rows[-9:]);kijun=midpoint(rows[-26:])
    prev_tenkan=midpoint(rows[-10:-1])
    old=rows[:-26]
    span_a=(midpoint(old[-9:])+midpoint(old[-26:]))/2
    span_b=midpoint(old[-52:])
    return tenkan,kijun,prev_tenkan,max(span_a,span_b)

def bottom_rebound_metrics(day,h4,h1,btc4,at,baseline,contiguous,bar_end):

class TargetedReplayRecoveryTest(unittest.TestCase):
    """Use only real returned 1h source bars; no 30m price substitution."""

    def setUp(self):
        self.at=datetime(2026,10,31,15,tzinfo=KST)
        self.base=[{"t":self.at+timedelta(hours=i),"o":100.0,"h":100.1,
                    "l":99.9,"c":100.0,"turnover":1000} for i in range(120)]
        self.gap=self.at+timedelta(hours=1)
        self.calls=[]

    def get(self,market,tf,to,count=100):
        self.calls.append((market,tf,to,count))
        if tf=="1h" and count==160:
            return [r for r in self.base if r["t"]!=self.gap]
        if tf=="1h":
            return [r for r in self.base if r["t"]==self.gap]
        raise AssertionError("Unexpected timeframe")

    def test_exact_hour_recovered_and_fee_unchanged(self):
        got=main.swing_backtest_exit("KRW-TEST",self.at,.04,getter=self.get)
        self.assertEqual(got["status"],"RESOLVED")
        self.assertEqual(got["minutes"],7200)
        self.assertEqual(got["audit"]["recovered_one_hour_candles"],1)
        self.assertEqual(got["audit"]["status_code"],"RECOVERED_1H")
        self.assertEqual(got["audit"]["hourly_recheck_calls"],1)
        self.assertEqual(self.calls[1][2],self.gap+timedelta(hours=1,minutes=1))
        self.assertAlmostEqual(got["net_pct"],-0.079968,places=4)

    def test_persistent_hourly_gap_keeps_unknown(self):
        def no_gap_return(market,tf,to,count=100):
            return [r for r in self.base if r["t"]!=self.gap]
        got=main.swing_backtest_exit("KRW-TEST",self.at,.04,getter=no_gap_return)
        self.assertEqual(got["status"],"UNKNOWN")
        self.assertIsNone(got["net_pct"])
        self.assertEqual(got["audit"]["status_code"],"SOURCE_1H_NOT_RETURNED")
        self.assertEqual(got["audit"]["first_unresolved_or_recovered_hour_kst"],"2026-10-31 16:00")

    def test_30m_bars_diagnose_but_never_create_1h_execution_price(self):
        def half(market,tf,to,count=100):
            if tf=="30m":return [{"t":self.gap+timedelta(minutes=30),"o":99,"h":100,"l":99,"c":99}]
            return [r for r in self.base if r["t"]!=self.gap]
        got=main.swing_backtest_exit("KRW-TEST",self.at,.04,getter=half,detailed=True)
        self.assertEqual(got["status"],"UNKNOWN")
        self.assertIsNone(got["net_pct"])
        self.assertEqual(got["audit"]["status_code"],"SOURCE_1H_MISSING_BUT_30M_EXISTS")
        self.assertEqual(got["audit"]["lower_timeframe_bars_in_missing_hour"],1)

    def test_no_1h_or_30m_cannot_claim_zero_trades(self):
        def no_gap(market,tf,to,count=100):
            if tf=="30m":return []
            return [r for r in self.base if r["t"]!=self.gap]
        got=main.swing_backtest_exit("KRW-TEST",self.at,.04,getter=no_gap,detailed=True)
        self.assertEqual(got["status"],"UNKNOWN")
        self.assertEqual(got["audit"]["status_code"],"SOURCE_1H_30M_NOT_RETURNED")

    def test_recheck_api_error_is_not_converted_to_profit(self):
        def broken(market,tf,to,count=100):
            if tf=="1h" and count!=160:raise RuntimeError("temporary HTTP 503")
            return [r for r in self.base if r["t"]!=self.gap]
        got=main.swing_backtest_exit("KRW-TEST",self.at,.04,getter=broken)
        self.assertEqual(got["status"],"UNKNOWN")
        self.assertEqual(got["audit"]["status_code"],"HOURLY_RECHECK_API_ERROR")
        self.assertIn("503",got["audit"]["detail_error"])

    def test_gap_after_stop_skips_unnecessary_recheck(self):
        late=self.at+timedelta(hours=65)
        def stops(market,tf,to,count=100):
            self.calls.append((tf,count))

        got=main.swing_backtest_exit("KRW-TEST",self.at,.04,getter=stops,detailed=True)

        self.assertEqual(got["audit"]["total_candle_api_calls"],1)

    def test_missing_entry_may_be_recovered_only_by_true_hourly_bar(self):
        def missing_first(market,tf,to,count=100):
            if tf=="1h" and count==160:return [r for r in self.base if r["t"]!=self.at]
            if tf=="1h":return [r for r in self.base if r["t"]==self.at]
            return []
        got=main.swing_backtest_exit("KRW-TEST",self.at,.04,getter=missing_first)
