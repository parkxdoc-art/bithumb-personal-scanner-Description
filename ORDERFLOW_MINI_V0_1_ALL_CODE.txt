"""Bithumb ORDERFLOW MINI V0.1: public REST snapshots, no trading.

Only five fixed KRW markets. One best-five-level orderbook snapshot and
recent actual trades per refresh. No claims of persistent walls or predictive BUY.
"""
from __future__ import annotations

from concurrent.futures import ThreadPoolExecutor, as_completed
from datetime import datetime
from threading import Lock
from zoneinfo import ZoneInfo
import math
import time

import requests

VERSION = "ORDERFLOW MINI V0.1"
MARKETS = ("KRW-XRP", "KRW-ETH", "KRW-NEAR", "KRW-SOL", "KRW-DOGE")
API = "https://api.bithumb.com"
KST = ZoneInfo("Asia/Seoul")
TRADE_WINDOW_SECONDS = 30
TRADE_COUNT = 200
CACHE_SECONDS = 8
_CACHE = {"at": 0.0, "data": None}
_LOCK = Lock()


def _request(path, params):
    """Read-only public API; no credentials or orders."""
    response = requests.get(
        API + path,
        params=params,
        headers={"Accept": "application/json", "User-Agent": "personal-orderflow-research/0.1"},
        timeout=(3, 7),
    )
    response.raise_for_status()
    data = response.json()
    if not isinstance(data, list):
        raise ValueError("Bithumb response was not an array")
    return data


def _number(x):
    try:
        n = float(x)
        return n if math.isfinite(n) and n >= 0 else None
    except (TypeError, ValueError, OverflowError):
        return None


def analyze(market, book, trades, now_ms, trade_error=None):
    """Pure function: distinguish resting orders from executed trades."""
    result = {
        "market": market, "version": VERSION,
        "book_ok": False, "trade_ok": trade_error is None,
        "error": None, "trade_error": trade_error,
        "book_timestamp_ms": None, "book_age_s": None,
        "bid5_krw": None, "ask5_krw": None, "bid_ask_ratio": None,
        "bid_share_pct": None, "best_bid": None, "best_ask": None,
        "spread_pct": None, "bids": [], "asks": [],
        "trade_count_30s": None, "trade_buy_krw_30s": None,
        "trade_sell_krw_30s": None, "trade_net_krw_30s": None,
        "trade_buy_share_pct_30s": None, "trade_total_krw_30s": None,
        "trade_sample_capped": False, "last_trade_age_s": None,
        "checks": {}, "score": None, "score_total": 4, "status": "데이터 확인",
    }
    if not isinstance(book, dict) or book.get("market") != market:
        result["error"] = "호가 응답 없음"
        return result
    units = book.get("orderbook_units") or []
    bids, asks = [], []
    for u in units:
        if not isinstance(u, dict):
            continue
        bp, ap = _number(u.get("bid_price")), _number(u.get("ask_price"))
        bs, a_s = _number(u.get("bid_size")), _number(u.get("ask_size"))
        if bp and bs is not None and bs > 0:
            bids.append({"price": bp, "size": bs, "krw": bp * bs})
        if ap and a_s is not None and a_s > 0:
            asks.append({"price": ap, "size": a_s, "krw": ap * a_s})
    bids = sorted(bids, key=lambda x: -x["price"])[:5]
    asks = sorted(asks, key=lambda x: x["price"])[:5]
    if not bids or not asks or asks[0]["price"] <= bids[0]["price"]:
        result["error"] = "유효한 매수·매도 1호가 없음"
        return result
    ts = _number(book.get("timestamp"))
    age = (now_ms - ts) / 1000 if ts is not None else None
    result.update(
        book_ok=True, book_timestamp_ms=int(ts) if ts is not None else None,
        book_age_s=round(age, 1) if age is not None else None,
        bids=bids, asks=asks, best_bid=bids[0]["price"], best_ask=asks[0]["price"],
    )
    bidkrw, askkrw = sum(x["krw"] for x in bids), sum(x["krw"] for x in asks)
    spread = 100 * (asks[0]["price"] - bids[0]["price"]) / (
        (asks[0]["price"] + bids[0]["price"]) / 2
    )
    result.update(
        bid5_krw=round(bidkrw), ask5_krw=round(askkrw),
        bid_ask_ratio=round(bidkrw / askkrw, 3) if askkrw else None,
        bid_share_pct=round(100 * bidkrw / (bidkrw + askkrw), 1) if bidkrw + askkrw else None,
        spread_pct=round(spread, 4),
    )
    fresh = age is not None and -5 <= age <= 30
    result["book_fresh"] = fresh
    if not fresh:
        result["error"] = "호가 갱신 시각이 오래됐거나 비정상적임"

    if trade_error is not None or not isinstance(trades, list):
        result["trade_ok"] = False
    else:
        eligible = []
        valid = []
        for t in trades:
            if not isinstance(t, dict):
                continue
            if t.get("market") not in (None, market):
                continue
            stamp = _number(t.get("timestamp"))
            price, volume = _number(t.get("trade_price")), _number(t.get("trade_volume"))
            side = t.get("ask_bid")
            if stamp is None or not price or not volume or side not in ("BID", "ASK"):
                continue
            # Do not use future trades or trades outside 30s.
            if stamp > now_ms + 1000 or stamp < now_ms - 120000:
                continue
            valid.append((stamp, side, price * volume))
            if now_ms - 30000 <= stamp <= now_ms + 1000:
                eligible.append((stamp, side, price * volume))
        if valid:
            result["last_trade_age_s"] = round((now_ms - max(x[0] for x in valid)) / 1000, 1)
        # If all 200 recent ticks occurred inside 30 seconds, REST truncated
        # the window. A ratio based on them is biased, so do not score it.
        capped = len(trades) >= TRADE_COUNT and valid and min(x[0] for x in valid) >= now_ms - 30000
        buys = sum(v for _, side, v in eligible if side == "BID")
        sells = sum(v for _, side, v in eligible if side == "ASK")
        total = buys + sells
        result.update(
            trade_sample_capped=bool(capped),
            trade_count_30s=len(eligible),
            trade_buy_krw_30s=round(buys),
            trade_sell_krw_30s=round(sells),
            trade_net_krw_30s=round(buys - sells),
            trade_total_krw_30s=round(total),
            trade_buy_share_pct_30s=round(100 * buys / total, 1) if total else None,
        )
    # Conditions are exploratory screen labels, NOT probability or BUY signals.
    # Never score stale books, API failures or incomplete trade windows.
    trade_valid = (
        result["trade_ok"] and not result["trade_sample_capped"]
        and result["trade_count_30s"] is not None
        and result["trade_count_30s"] >= 3
    )
    checks = {
        "근접 5호가 매수대금 우세(1.25배 이상)": bool(fresh and result["bid_ask_ratio"] is not None and result["bid_ask_ratio"] >= 1.25),
        "최근 30초 매수 체결 비중(58% 이상)": bool(trade_valid and result["trade_buy_share_pct_30s"] is not None and result["trade_buy_share_pct_30s"] >= 58),
        "호가 스프레드(0.20% 이하)": bool(fresh and spread <= 0.20),
        "최근 30초 체결대금(1천만원 이상)": bool(trade_valid and result["trade_total_krw_30s"] >= 10000000),
    }
    result["checks"] = checks
    if not fresh or not result["trade_ok"] or result["trade_sample_capped"] or not trade_valid:
        result["status"] = "자료 주의"
        # A partial score would give a false impression of 0/4.
        result["score"] = None
    else:
        result["score"] = sum(checks.values())
        result["status"] = "관찰 우세" if result["score"] == 4 else (
            "추가 관찰" if result["score"] >= 2 else "관망"
        )
    return result


def scan(fetcher=None, now_ms=None):
    """One snapshot. All five markets use the same server time cutoff."""
    fetcher = fetcher or _request
    if now_ms is None:
        now_ms = int(time.time() * 1000)
    books = fetcher("/v1/orderbook", {"markets": ",".join(MARKETS)})
    book_by_market = {
        x.get("market"): x for x in books if isinstance(x, dict) and x.get("market") in MARKETS
    }
    trade_by_market, errors = {}, {}
    def get_trades(market):
        return fetcher("/v1/trades/ticks", {"market": market, "count": TRADE_COUNT})
    with ThreadPoolExecutor(max_workers=3) as pool:
        futures = {pool.submit(get_trades, m): m for m in MARKETS}
        for future in as_completed(futures):
            market = futures[future]
            try:
                trade_by_market[market] = future.result()
            except Exception as exc:
                errors[market] = str(exc)[:150]
    items = [
        analyze(m, book_by_market.get(m), trade_by_market.get(m), now_ms, errors.get(m))
        for m in MARKETS
    ]
    return {
        "version": VERSION, "source": "Bithumb public REST API",
        "fetched_at_ms": now_ms, "refresh_interval_seconds": 15,
        "markets": items, "count": len(items),
        "notes": [
            "근접 5호가 금액은 현재 대기 잔량이다. 실제 매수 체결로 해석하면 안 된다.",
            "매수·매도 체결은 최근 30초의 REST 표본이며 WebSocket 누적 CVD가 아니다.",
            "매수벽 유지·주문 취소 여부는 단일 호가 스냅샷으로 판단할 수 없다.",
            "4/4 점수는 임의 설정한 관찰조건 충족 개수이지 수익 확률/BUY 신호가 아니다.",
            "최소 체결대금, 수수료, 슬리피지, 호가 깊이는 실전 적합성을 보장하지 않는다.",
        ],
    }


def cached_scan():
    """Prevent unnecessary repeat public-API requests across page viewers."""
    with _LOCK:
        elapsed = time.monotonic() - _CACHE["at"]
        if _CACHE["data"] is not None and 0 <= elapsed < CACHE_SECONDS:
            return {**_CACHE["data"], "cached": True}
        payload = scan()
        _CACHE.update(at=time.monotonic(), data=payload)
        return {**payload, "cached": False}

class OrderflowMiniV01ResearchTest(unittest.TestCase):
    def setUp(self):
        import orderflow_mini_lab as flow
        self.flow=flow
        self.now=1800000000000

    def book(self,market="KRW-XRP",age_ms=1000):
        return {"market":market,"timestamp":self.now-age_ms,
            "orderbook_units":[{"bid_price":100,"bid_size":1000000,
                                "ask_price":100.1,"ask_size":100000}]}

    def ticks(self,market="KRW-XRP"):
        return ([{"market":market,"timestamp":self.now-2000,
                  "trade_price":100,"trade_volume":50000,"ask_bid":"BID"}]*4+
                [{"market":market,"timestamp":self.now-3000,
                  "trade_price":100,"trade_volume":25000,"ask_bid":"ASK"}]*2)

    def test_book_notional_and_actual_trade_are_separate(self):
        r=self.flow.analyze("KRW-XRP",self.book(),self.ticks(),self.now)
        self.assertTrue(r["book_ok"])
        self.assertEqual(r["bid5_krw"],100000000)
        self.assertEqual(r["ask5_krw"],10010000)
        self.assertEqual(r["trade_buy_krw_30s"],20000000)
        self.assertEqual(r["trade_sell_krw_30s"],5000000)
        self.assertEqual(r["trade_net_krw_30s"],15000000)
        self.assertEqual(r["trade_buy_share_pct_30s"],80)
        self.assertEqual(r["score"],4)
        self.assertEqual(r["status"],"관찰 우세")
        self.assertEqual(r["trade_count_30s"],6)

    def test_stale_orderbook_is_not_a_signal(self):
        r=self.flow.analyze("KRW-XRP",self.book(age_ms=90000),
                            self.ticks(),self.now)
        self.assertIsNone(r["score"])
        self.assertFalse(r["book_fresh"])
        self.assertEqual(r["status"],"자료 주의")

    def test_rest_trade_sample_capped_blocks_scoring(self):
        many=self.ticks()[:1]*self.flow.TRADE_COUNT
        r=self.flow.analyze("KRW-XRP",self.book(),many,self.now)
        self.assertTrue(r["trade_sample_capped"])
        self.assertIsNone(r["score"])

    def test_no_recent_trades_is_not_fabricated_buy_or_zero_score(self):
        old=[{"market":"KRW-XRP","timestamp":self.now-60000,
              "trade_price":100,"trade_volume":200,
              "ask_bid":"BID"}]
        r=self.flow.analyze("KRW-XRP",self.book(),old,self.now)
        self.assertEqual(r["trade_count_30s"],0)
        self.assertIsNone(r["trade_buy_share_pct_30s"])
        self.assertIsNone(r["score"])
        self.assertEqual(r["status"],"자료 주의")

    def test_trade_error_keeps_book_notional_but_no_score(self):
        r=self.flow.analyze("KRW-XRP",self.book(),None,self.now,
                            trade_error="429 Too Many Requests")
        self.assertTrue(r["book_ok"])
        self.assertEqual(r["bid5_krw"],100000000)
        self.assertIsNone(r["score"])
        self.assertIn("429",r["trade_error"])

    def test_all_five_fixed_markets_scan_one_book_and_trade_each(self):
        calls=[]
        def fetch(path,params):
            calls.append((path,params))
            if path=="/v1/orderbook":
                return [self.book(m) for m in self.flow.MARKETS]
            return self.ticks(params["market"])
        out=self.flow.scan(fetcher=fetch,now_ms=self.now)
        self.assertEqual(out["version"],"ORDERFLOW MINI V0.1")
        self.assertEqual(len(out["markets"]),5)
        self.assertEqual(len(calls),6)
        self.assertEqual(sum(path=="/v1/orderbook" for path,_ in calls),1)
        self.assertEqual(len(set(x["market"] for x in out["markets"])),5)
        self.assertTrue(all(x["score"]==4 for x in out["markets"]))
        self.assertTrue(all(x["trade_count_30s"]==6 for x in out["markets"]))

    def test_bad_orderbook_cannot_create_false_candidate(self):
        r=self.flow.analyze("KRW-XRP",
                            {"market":"KRW-XRP","orderbook_units":[]},
                            self.ticks(),self.now)
        self.assertFalse(r["book_ok"])
        self.assertIsNone(r["score"])
        self.assertEqual(r["status"],"데이터 확인")
